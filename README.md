    function highlightNode(metricName, metricValue) {
        const terminal = document.getElementById('terminalStream');
        const newLog = document.createElement('div');
        newLog.className = 'text-sky-200';
        newLog.innerHTML = `> TELEMETRY: [${metricName.toUpperCase()}] = ${metricValue}`;
        terminal.prepend(newLog);
        
        // Limit log items
        if(terminal.children.length > 5) {
            terminal.removeChild(terminal.lastChild);
        }
    }
